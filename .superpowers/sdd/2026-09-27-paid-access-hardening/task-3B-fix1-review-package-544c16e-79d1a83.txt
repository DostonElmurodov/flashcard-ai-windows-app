+    private var firstReply: ((Result<(Int, Data), Error>) -> Void)?
+    private var count = 0

+    func receive(_ reply: @escaping (Result<(Int, Data), Error>) -> Void,
+                 firstArrived: () -> Void, laterBody: Data) {
+        lock.lock()
+        count += 1
+        let isFirst = count == 1
+        if isFirst { firstReply = reply }
+        lock.unlock()
+        if isFirst { firstArrived() }
+        else { reply(.success((200, laterBody))) }

+        let reply = firstReply

+    }

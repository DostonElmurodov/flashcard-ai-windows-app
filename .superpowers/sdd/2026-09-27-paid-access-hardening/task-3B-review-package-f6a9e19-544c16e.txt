+private final class RestoreProofURLProtocol: URLProtocol {
+    static var respond: ((URLRequest) throws -> (Int, Data))?
+
+    override class func canInit(with request: URLRequest) -> Bool { true }
+    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }
+    override func startLoading() {
+        do {
+            guard let respond = Self.respond else { throw URLError(.badServerResponse) }
+            let (status, body) = try respond(request)
+            let response = HTTPURLResponse(url: request.url!, statusCode: status, httpVersion: nil, headerFields: nil)!
+            client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
+            client?.urlProtocol(self, didLoad: body)
+            client?.urlProtocolDidFinishLoading(self)
+        } catch { client?.urlProtocol(self, didFailWithError: error) }
+    }
+    override func stopLoading() {}
+}

+private final class FakeRestoreEvidenceSource: StoreKitPurchaseEvidenceSource {
+    var current: [ApplePurchaseEvidence] = []
+    var historical: [ApplePurchaseEvidence] = []
+    var currentError: Error?
+    var historyError: Error?
+    var syncCount = 0
+    var historicalProductIds: [String] = []
+    var finished: [String] = []
+    var currentOperation: (() async throws -> [ApplePurchaseEvidence])?
+
+    func currentEvidence() async throws -> [ApplePurchaseEvidence] {
+        if let currentOperation { return try await currentOperation() }
+        if let currentError { throw currentError }
+        return current
+    }
+    func latestHistoricalEvidence(productIds: [String]) async throws -> [ApplePurchaseEvidence] {
+        historicalProductIds = productIds
+        if let historyError { throw historyError }
+        return historical
+    }
+    func synchronize() async throws { syncCount += 1 }
+    func finish(transactionId: String) async { finished.append(transactionId) }
+}

+        if let body = request.httpBody { return body }
+        guard let stream = request.httpBodyStream else { return Data() }
+        stream.open()
+        defer { stream.close() }
+        var body = Data()
+        var buffer = [UInt8](repeating: 0, count: 1024)
+        while stream.hasBytesAvailable {
+            let count = stream.read(&buffer, maxLength: buffer.count)
+            if count <= 0 { break }
+            body.append(buffer, count: count)
+        }
+        return body

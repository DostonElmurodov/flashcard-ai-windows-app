+private final class MutationNoNetworkProtocol: URLProtocol {
+    static var responseData: Data?
+    static var requestCount = 0
+    override class func canInit(with request: URLRequest) -> Bool { true }
+    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }
+    override func startLoading() {
+        Self.requestCount += 1
+        if let data = Self.responseData, let url = request.url, url.host == "example.invalid",
+           url.path == "/owlai/ai/word-detail", request.httpMethod == "POST" {
+            client?.urlProtocol(self, didReceive: HTTPURLResponse(url: url, statusCode: 200, httpVersion: nil, headerFields: nil)!, cacheStoragePolicy: .notAllowed)
+            client?.urlProtocol(self, didLoad: data)
+            client?.urlProtocolDidFinishLoading(self)
+            return
+        }
+        XCTFail("Entitlement mutation tests must not call a network service")
+        client?.urlProtocol(self, didFailWithError: URLError(.unsupportedURL))
+    }
+    override func stopLoading() {}

+private final class MutationOneShot: @unchecked Sendable {
+    private let lock = NSLock()
+    private var hasRun = false

+        hasRun = true
+        lock.unlock()

+    }

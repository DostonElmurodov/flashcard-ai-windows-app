+    public async Task<AppAttestProofResult> VerifyAsync(
+        string keyId, byte[] rawBody, string path, string challengeId,
+        string assertionBase64, CancellationToken ct)
+    {
+        if (string.IsNullOrWhiteSpace(keyId) || string.IsNullOrWhiteSpace(challengeId)
+            || string.IsNullOrWhiteSpace(assertionBase64))
+            return AppAttestProofResult.Invalid;
+
+        var nonce = await challenges.TryConsumeAsync(challengeId,
+            ChallengeService.ChallengeKind.Assertion, keyId, path, ct);
+        if (nonce is null) return AppAttestProofResult.Invalid;
+
+        byte[] assertion;
+        try { assertion = Convert.FromBase64String(assertionBase64); }
+        catch (FormatException) { return AppAttestProofResult.Invalid; }
+
+        var device = await db.Devices.AsNoTracking().SingleOrDefaultAsync(d => d.KeyId == keyId, ct);
+        if (device is null || device.PublicKey.Length == 0) return AppAttestProofResult.Invalid;
+
+        var hash = AppAttestClientData.ComputeAssertionHash(nonce, rawBody, path, challengeId);
+        var verified = verifier.VerifyAssertion(assertion, device.PublicKey, device.SignCount, hash);
+        if (!verified.Ok || verified.NewSignCount <= device.SignCount)
+            return AppAttestProofResult.Invalid;
+
+        var now = timeProvider.GetUtcNow().UtcDateTime;
+        var changed = await db.Devices
+            .Where(d => d.KeyId == keyId && d.SignCount == device.SignCount)
+            .ExecuteUpdateAsync(s => s
+                .SetProperty(d => d.SignCount, verified.NewSignCount)
+                .SetProperty(d => d.LastSeenAt, now), ct);
+        return changed == 1 ? AppAttestProofResult.Success : AppAttestProofResult.Invalid;

+    private static byte[] ManualHash(byte[] nonce, byte[] body, string path, string challengeId)
+    {
+        using var bytes = new MemoryStream();
+        bytes.Write(nonce);
+        bytes.WriteByte(0x1f);
+        bytes.Write(SHA256.HashData(body));
+        bytes.WriteByte(0x1f);
+        bytes.Write(Encoding.UTF8.GetBytes(path));
+        bytes.WriteByte(0x1f);
+        bytes.Write(Encoding.UTF8.GetBytes(challengeId));
+        return SHA256.HashData(bytes.ToArray());
+    }

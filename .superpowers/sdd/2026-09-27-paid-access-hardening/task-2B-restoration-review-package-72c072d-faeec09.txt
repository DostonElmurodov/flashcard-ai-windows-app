+using System.Text.Json;
+
+namespace Mavrylo.Filters;
+
+internal static class PurchaseRequestFields
+{
+    public static bool ValidForPath(PathString path, byte[] rawBody)
+        => path.Value?.TrimEnd('/').ToLowerInvariant() switch
+        {
+            "/owlai/iap/verify" => Valid(rawBody, 256_000, ["jws_transaction"], ["device_uuid"]),
+            "/owlai/iap/purchase-context" => Valid(rawBody, 16_000, ["product_id"], []),
+            "/owlai/iap/restore" => Valid(rawBody, 80_000,
+                ["jws_transaction", "jws_app_transaction", "device_verification_id"], []),
+            "/owlai/account/subscription/apple/claim" => Valid(rawBody, 40_000,
+                ["account_id", "jws_transaction"], []),
+            _ => true
+        };
+
+    private static bool Valid(byte[] rawBody, int maxBytes, string[] required, string[] optional)
+    {
+        if (rawBody.Length > maxBytes) return false;
+        try
+        {
+            using var doc = JsonDocument.Parse(rawBody);
+            if (doc.RootElement.ValueKind != JsonValueKind.Object) return false;
+            var allowed = required.Concat(optional).ToHashSet(StringComparer.Ordinal);
+            var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
+            foreach (var property in doc.RootElement.EnumerateObject())
+            {
+                if (!allowed.Contains(property.Name) || !seen.Add(property.Name)) return false;
+                if (property.Value.ValueKind != JsonValueKind.String
+                    && !(optional.Contains(property.Name, StringComparer.Ordinal)
+                        && property.Value.ValueKind == JsonValueKind.Null)) return false;
+            }
+            return required.All(seen.Contains);
+        }
+        catch (JsonException) { return false; }
+    }
+}

+public sealed class MobilePurchaseAccessResolver(AppDbContext db, EntitlementService entitlements, TimeProvider? clock = null)
+{
+    private sealed record Candidate(SubscriptionEntity Purchase, Guid OwnerId, string Authority, DateTime? GrantUntil,
+        EntitlementService.Entitlement Entitlement);
+
+    public async Task<MobilePurchaseAccess?> ResolveAsync(string keyId, CancellationToken ct)
+    {
+        var device = await db.Devices.AsNoTracking().SingleOrDefaultAsync(x => x.KeyId == keyId, ct);
+        if (device == null) return null;
+        var candidates = new List<Candidate>();
+        if (device.AnonymousOwnerId is { } installationOwner)
+        {
+            var bindings = await db.ApplePurchaseBindings.AsNoTracking()
+                .Where(x => x.AnonymousOwnerId == installationOwner && x.BindingKind == "server_token")
+                .ToListAsync(ct);
+            foreach (var binding in bindings)
+                await AddAsync(binding, "owner", null);
+        }
+        var grants = await db.MobilePurchaseGrants.AsNoTracking().Where(x => x.DeviceId == device.Id).ToListAsync(ct);
+        foreach (var grant in grants)
+        {
+            var binding = await db.ApplePurchaseBindings.AsNoTracking().SingleOrDefaultAsync(x =>
+                x.Environment == grant.Environment && x.OriginalTransactionId == grant.OriginalTransactionId, ct);
+            if (binding != null) await AddAsync(binding, "mobile_restore", grant.ValidUntil);
+        }
+
+        var selected = candidates.OrderByDescending(x => EntitlementService.Rank(x.Entitlement.Status))
+            .ThenByDescending(x => x.Entitlement.ExpiresAt).FirstOrDefault();
+        var stickyPaid = candidates.Any(x => entitlements.IsApplicable(x.Purchase) && x.Purchase.WasEverPaid);
+        return selected == null
+            ? new MobilePurchaseAccess(device.Id, device.AnonymousOwnerId, null, null, "none", null,
+                EntitlementService.Entitlement.Free())
+            : new MobilePurchaseAccess(device.Id, device.AnonymousOwnerId, selected.OwnerId, selected.Purchase,
+                selected.Authority, selected.GrantUntil, selected.Entitlement with { WasEverPaid = stickyPaid });
+
+        async Task AddAsync(ApplePurchaseBindingEntity binding, string authority, DateTime? grantUntil)
+        {
+            var purchase = await db.Subscriptions.AsNoTracking().SingleOrDefaultAsync(x =>
+                x.OriginalTransactionId == binding.OriginalTransactionId && x.Environment == binding.Environment, ct);
